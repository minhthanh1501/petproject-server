    USERS {
        bigint id PK
        string full_name
        string email
        string phone
        int reward_points
    }

    ROLES {
        int id PK
        string code
        string name
    }

    PERMISSIONS {
        int id PK
        string code
        string module
    }

    USER_ROLES {
        bigint user_id PK,FK
        int role_id PK,FK
        int branch_id FK
    }

    ROLE_PERMISSIONS {
        int role_id PK,FK
        int permission_id PK,FK
    }

    BOOTHS {
        int id PK
        int area_id FK
        string code
        string type
        string status
    }

    STUDY_SESSIONS {
        bigint id PK
        bigint user_id FK
        int booth_id FK
        timestamp check_in_time
        string status
    }

    INVOICES {
        bigint id PK
        bigint session_id FK
        decimal total_amount
        string payment_status
    }
